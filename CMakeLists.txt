function(walnut_link_items item out)
    set(items "")
    if(TARGET ${item})
        get_target_property(real ${item} ALIASED_TARGET)
        if(real)
            set(item ${real})
        endif()
        get_target_property(type ${item} TYPE)
        get_target_property(imported ${item} IMPORTED)
        if(imported AND NOT type STREQUAL "INTERFACE_LIBRARY")
            get_target_property(loc ${item} IMPORTED_LOCATION)
            if(NOT loc)
                get_target_property(configs ${item} IMPORTED_CONFIGURATIONS)
                if(configs)
                    foreach(c IN LISTS configs)
                        get_target_property(loc ${item} IMPORTED_LOCATION_${c})
                        if(loc)
                            break()
                        endif()
                    endforeach()
                endif()
            endif()
            if(loc)
                list(APPEND items "${loc}")
            endif()
        endif()
        get_target_property(deps ${item} INTERFACE_LINK_LIBRARIES)
        if(deps)
            foreach(d IN LISTS deps)
                walnut_link_items("${d}" sub)
                list(APPEND items ${sub})
            endforeach()
        endif()
    elseif(item MATCHES "^\\$<")
    elseif(item MATCHES "^-" OR IS_ABSOLUTE "${item}")
        list(APPEND items "${item}")
    elseif(item)
        list(APPEND items "-l${item}")
    endif()
    set(${out} "${items}" PARENT_SCOPE)
endfunction()

get_target_property(WALNUT_FIZMO_TYPE fizmo TYPE)
if(WALNUT_FIZMO_TYPE STREQUAL "SHARED_LIBRARY" AND NOT WIN32)
    list(APPEND WALNUT_RUNTIME_LIB_LIST "-Wl,-rpath,$<TARGET_FILE_DIR:fizmo>")
endif()
include("${fizmo_SOURCE_DIR}/Fizmo/cmake/fizmoDependencies.cmake")
foreach(dep IN LISTS FIZMO_DEPENDENCY_TARGETS)
    walnut_link_items("${dep}" WALNUT_DEP_ITEMS)
    list(APPEND WALNUT_RUNTIME_LIB_LIST ${WALNUT_DEP_ITEMS})
endforeach()
if(UNIX)
    list(APPEND WALNUT_RUNTIME_LIB_LIST "-pthread")
endif()
list(REMOVE_DUPLICATES WALNUT_RUNTIME_LIB_LIST)
list(JOIN WALNUT_RUNTIME_LIB_LIST ";" WALNUT_RUNTIME_LIBS)

enable_testing()

file(GLOB WALNUT_E2E_CASES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/e2e/*.wal")
foreach(case ${WALNUT_E2E_CASES})
    get_filename_component(case_name "${case}" NAME_WE)
    add_test(NAME e2e.${case_name}
        COMMAND ${CMAKE_COMMAND} -DWALNUT=$<TARGET_FILE:walnut> -DCASE=${case} -DWORK=${CMAKE_CURRENT_BINARY_DIR}/test-work
                -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_e2e.cmake")
    set_tests_properties(e2e.${case_name} PROPERTIES TIMEOUT 300)
endforeach()

file(GLOB WALNUT_SEMANTIC_SUITES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/semantic/*.txt")
foreach(suite ${WALNUT_SEMANTIC_SUITES})
    get_filename_component(suite_name "${suite}" NAME_WE)
    add_test(NAME semantic.${suite_name}
        COMMAND ${CMAKE_COMMAND} -DWALNUT=$<TARGET_FILE:walnut> -DSUITE=${suite} -DWORK=${CMAKE_CURRENT_BINARY_DIR}/test-work
                -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_semantic.cmake")
    set_tests_properties(semantic.${suite_name} PROPERTIES TIMEOUT 900)
endforeach()
