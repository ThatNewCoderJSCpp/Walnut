function(walnut_link_items item out)
    set(items "")
    if(TARGET ${item})
        get_target_property(real ${item} ALIASED_TARGET)
        if(real)
            set(item ${real})
        endif()
        get_target_property(type ${item} TYPE)
        get_target_property(imported ${item} IMPORTED)
        if(imported AND NOT type STREQUAL "INTERFACE_LIBRARY")
            get_target_property(loc ${item} IMPORTED_LOCATION)
            if(NOT loc)
                get_target_property(configs ${item} IMPORTED_CONFIGURATIONS)
                if(configs)
                    foreach(c IN LISTS configs)
                        get_target_property(loc ${item} IMPORTED_LOCATION_${c})
                        if(loc)
                            break()
                        endif()
                    endforeach()
                endif()
            endif()
            if(loc)
                list(APPEND items "${loc}")
            endif()
        endif()
        get_target_property(deps ${item} INTERFACE_LINK_LIBRARIES)
        if(deps)
            foreach(d IN LISTS deps)
                walnut_link_items("${d}" sub)
                list(APPEND items ${sub})
            endforeach()
        endif()
    elseif(item MATCHES "^\\$<")
    elseif(item MATCHES "^-" OR IS_ABSOLUTE "${item}")
        list(APPEND items "${item}")
    elseif(item)
        list(APPEND items "-l${item}")
    endif()
    set(${out} "${items}" PARENT_SCOPE)
endfunction()

function(walnut_portable_libs in_list out)
    set(result "")
    foreach(item IN LISTS ${in_list})
        if(IS_ABSOLUTE "${item}" AND item MATCHES "/lib([^/]+)\\.dll\\.a$")
            list(APPEND result "-l${CMAKE_MATCH_1}")
        elseif(IS_ABSOLUTE "${item}" AND item MATCHES "/lib([^/]+)\\.(so|a|lib)(\\.[0-9.]+)?$")
            list(APPEND result "-l${CMAKE_MATCH_1}")
        else()
            list(APPEND result "${item}")
        endif()
    endforeach()
    set(${out} "${result}" PARENT_SCOPE)
endfunction()

set(_walnut_library_entries "")
set(_walnut_library_index 0)
foreach(lib_file IN LISTS WALNUT_LIBRARY_FILES)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${lib_file}")
    file(RELATIVE_PATH lib_rel "${CMAKE_CURRENT_SOURCE_DIR}/Library" "${lib_file}")
    file(READ "${lib_file}" lib_hex HEX)
    string(LENGTH "${lib_hex}" lib_hex_len)
    math(EXPR lib_size "${lib_hex_len} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," lib_bytes "${lib_hex}")
    string(APPEND _walnut_library "inline const unsigned char file_${_walnut_library_index}[] = { ${lib_bytes} 0x00 };\n")
    string(APPEND _walnut_library_entries "    { \"${lib_rel}\", file_${_walnut_library_index}, ${lib_size} },\n")
    math(EXPR _walnut_library_index "${_walnut_library_index} + 1")
endforeach()
string(APPEND _walnut_library "\ninline const File files[] = {\n${_walnut_library_entries}    { nullptr, nullptr, 0 }\n};\n\n} // namespace walnut_library\n\n#endif // WALNUT_LIBRARY_HPP\n")

        set(_walnut_arch arm64)
