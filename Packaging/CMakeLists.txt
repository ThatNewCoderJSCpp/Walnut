cmake_minimum_required(VERSION 3.16)
project(walnut_packaging NONE)

set(WALNUT_STAGE "" CACHE PATH "Folder holding an unpacked Walnut package (bin, runtime, docs, ...)")
set(WALNUT_VERSION "" CACHE STRING "Walnut version used in package file names")
set(WALNUT_PLATFORM "" CACHE STRING "Platform name used in package file names, like linux-x86_64 or windows-ucrt64")
set(WALNUT_PACKAGE_SUFFIX "" CACHE STRING "Text put after the platform in package file names, like -with-compiler")
set(WALNUT_GENERATORS "TXZ" CACHE STRING "Package formats cpack produces")

foreach(v STAGE VERSION PLATFORM)
    if(NOT WALNUT_${v})
        message(FATAL_ERROR "Pass -DWALNUT_${v}=...")
    endif()
endforeach()
if(NOT EXISTS "${WALNUT_STAGE}/bin")
    message(FATAL_ERROR "${WALNUT_STAGE} doesn't hold a Walnut package folder")
endif()

if(WALNUT_PLATFORM MATCHES "arm64")
    set(_walnut_arch arm64)
elseif(WALNUT_PLATFORM MATCHES "i686")
    set(_walnut_arch i686)
else()
    set(_walnut_arch x86_64)
endif()

install(DIRECTORY "${WALNUT_STAGE}/" DESTINATION . USE_SOURCE_PERMISSIONS)

set(CPACK_PACKAGE_NAME Walnut)
set(CPACK_PACKAGE_VENDOR ThatNewCoderJSCpp)
set(CPACK_PACKAGE_CONTACT "ThatNewCoderJSCpp <https://github.com/ThatNewCoderJSCpp/Walnut>")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/ThatNewCoderJSCpp/Walnut")
set(CPACK_PACKAGE_VERSION ${WALNUT_VERSION})
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Walnut: a programming language and its compiler")
set(CPACK_PACKAGE_DESCRIPTION "Walnut is a programming language. This package has the walnut compiler, its runtime, the Fizmo libraries it builds against and its documentation.")
set(CPACK_PACKAGE_FILE_NAME Walnut-${WALNUT_VERSION}-${WALNUT_PLATFORM}${WALNUT_PACKAGE_SUFFIX})
set(CPACK_GENERATOR ${WALNUT_GENERATORS})
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
set(CPACK_STRIP_FILES OFF)
set(CPACK_THREADS 0)
set(CPACK_PROJECT_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/CPackOptions.cmake")

set(_walnut_scripts "${CMAKE_BINARY_DIR}/package-scripts")
file(REMOVE_RECURSE "${_walnut_scripts}")
file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/linux/deb-postinst" "${CMAKE_CURRENT_SOURCE_DIR}/linux/deb-prerm"
    DESTINATION "${_walnut_scripts}"
    FILE_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
file(MAKE_DIRECTORY "${_walnut_scripts}/deb")
file(RENAME "${_walnut_scripts}/deb-postinst" "${_walnut_scripts}/deb/postinst")
file(RENAME "${_walnut_scripts}/deb-prerm" "${_walnut_scripts}/deb/prerm")

set(CPACK_DEBIAN_PACKAGE_NAME walnut)
set(CPACK_DEBIAN_FILE_NAME ${CPACK_PACKAGE_FILE_NAME}.deb)
set(CPACK_DEBIAN_PACKAGE_SECTION devel)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "g++ | clang")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${_walnut_scripts}/deb/postinst;${_walnut_scripts}/deb/prerm")

set(CPACK_RPM_PACKAGE_NAME walnut)
set(CPACK_RPM_FILE_NAME ${CPACK_PACKAGE_FILE_NAME}.rpm)
set(CPACK_RPM_PACKAGE_GROUP "Development/Languages")
set(CPACK_RPM_PACKAGE_LICENSE "See https://github.com/ThatNewCoderJSCpp/Walnut")
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
set(CPACK_RPM_PACKAGE_AUTOPROV OFF)
set(CPACK_RPM_PACKAGE_SUGGESTS "gcc-c++")
set(CPACK_RPM_SPEC_MORE_DEFINE "%define _build_id_links none")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /opt)
set(CPACK_RPM_POST_INSTALL_SCRIPT_FILE "${CMAKE_CURRENT_SOURCE_DIR}/linux/rpm-post.sh")
set(CPACK_RPM_PRE_UNINSTALL_SCRIPT_FILE "${CMAKE_CURRENT_SOURCE_DIR}/linux/rpm-preun.sh")
if(_walnut_arch STREQUAL "arm64")
    set(CPACK_RPM_PACKAGE_ARCHITECTURE aarch64)
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
elseif(_walnut_arch STREQUAL "i686")
    set(CPACK_RPM_PACKAGE_ARCHITECTURE i686)
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE i386)
else()
    set(CPACK_RPM_PACKAGE_ARCHITECTURE x86_64)
    set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE amd64)
endif()

set(CPACK_NSIS_PACKAGE_NAME "Walnut ${WALNUT_VERSION}")
set(CPACK_NSIS_DISPLAY_NAME "Walnut ${WALNUT_VERSION}")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "Walnut")
set(CPACK_PACKAGE_INSTALL_REGISTRY_KEY "Walnut")
set(CPACK_NSIS_URL_INFO_ABOUT "https://github.com/ThatNewCoderJSCpp/Walnut")
set(CPACK_NSIS_MODIFY_PATH ON)
set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
if(_walnut_arch STREQUAL "i686")
    set(CPACK_NSIS_INSTALL_ROOT "$PROGRAMFILES")
else()
    set(CPACK_NSIS_INSTALL_ROOT "$PROGRAMFILES64")
endif()
set(CPACK_NSIS_EXTRA_INSTALL_COMMANDS "
    WriteRegStr SHCTX 'Software\\\\Microsoft\\\\Windows\\\\CurrentVersion\\\\App Paths\\\\walnut.exe' '' '$INSTDIR\\\\bin\\\\walnut.exe'
    WriteRegStr SHCTX 'Software\\\\Microsoft\\\\Windows\\\\CurrentVersion\\\\App Paths\\\\walnut.exe' 'Path' '$INSTDIR\\\\bin'
")
set(CPACK_NSIS_EXTRA_UNINSTALL_COMMANDS "
    DeleteRegKey SHCTX 'Software\\\\Microsoft\\\\Windows\\\\CurrentVersion\\\\App Paths\\\\walnut.exe'
")

include(CPack)
